# Star Fighter keyboard controls and settings, read from this file when the
# game starts.
#
#   CONTROL = key, key ...
#
# CONTROL is one of the 3DO pad controls below.  Keys are named as SDL names
# them, in any case: Up, Down, Left, Right, A..Z, 0..9, Return, Space, Escape,
# Backspace, Tab, Left Ctrl, Right Ctrl, Left Shift, Right Shift, Left Alt,
# Right Alt, Keypad 0..9, Keypad Enter, F1..F12, Comma (for ','), . / ; ' [ ] - =
# A control with nothing after '=' has no keys.  A control left out of this
# file keeps its built-in keys.  Delete this file to get the defaults back.
#
# What A, B, C, START, X, LS and RS do in flight is set in the game, under
# Configure > Setup Controls; the notes below give the default setup.
# A game controller works as well and is not affected by this file.

# pad up: menus up; in flight, nose down
UP    = K

# pad down: menus down; in flight, nose up
DOWN  = M

# pad left: roll left
LEFT  = Z

# pad right: roll right
RIGHT = X

# select; in flight (default setup) thrust
A     = Space

# in flight (default setup) ECM
B     = S

# in flight (default setup) change camera
C     = D

# start; in flight (default setup) map
START = J

# stop / back; in flight (default setup) pause menu
X     = P

# left shift; in flight (default setup) select weapon
LS    = A

# right shift; in flight (default setup) fire
RS    = L

# Display scale: the game is drawn at 320x240 times this number (1 to 4;
# 2 = 640x480).  The 3D view gets sharper; the HUD, menus, text and videos
# keep their size and look.  Higher scales need a faster computer.
SCALE = 1

# Draw distances, in map squares (the map is 256 x 256 squares and wraps round).
# The landscape is drawn in full detail within LAND_NEAR (4-32, multiple of 4),
# in medium detail within LAND_MEDIUM (8-64, multiple of 8) and with hills and
# mountains within LAND_FAR (16-128, multiple of 8); beyond that it is drawn flat.
# Each must be larger than the one before.  Larger distances need a faster computer.
LAND_NEAR = 16
LAND_MEDIUM = 32
LAND_FAR = 64

# Camera altitude, in map squares (1-127), above which each level of landscape
# detail is no longer drawn (each no lower than the one before).
LAND_NEAR_ALTITUDE = 32
LAND_MEDIUM_ALTITUDE = 48
LAND_FAR_ALTITUDE = 64

# 3D objects (16-120, multiple of 8): big ships, smoke, lasers and debris are drawn
# within OBJECT_DISTANCE; fighters, ground vehicles and buildings within half of it
# (fighters further away as small sprites).  Lock-on and the game itself are not
# affected.
OBJECT_DISTANCE = 64

# Camera altitude, in map squares (1-127), above which buildings and other ground
# objects are no longer drawn.  Lock-on is not affected.
BUILDING_ALTITUDE = 40
